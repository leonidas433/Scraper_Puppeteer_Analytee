"""
Database Models - SQLAlchemy ORM Definitions

This module defines all database tables and relationships for the analytics pipeline.
Supports both SQLite (development) and PostgreSQL (production).

Tables:
- places: Restaurant/business metadata
- reviews: Individual review records
- owner_responses: Owner responses to reviews
- kpi_summary: Aggregated KPIs by period
- analysis_cache: Performance caching layer
"""

from datetime import datetime, timedelta
from decimal import Decimal
from typing import Optional, List
from enum import Enum

from sqlalchemy import (
    Column, String, Integer, Float, Text, DateTime, 
    ForeignKey, Boolean, JSON, Index, UniqueConstraint,
    create_engine, event
)
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import relationship, Session
from sqlalchemy.pool import StaticPool

Base = declarative_base()


class Places(Base):
    """
    Tabla de locales/negocios.
    
    Almacena información general del negocio scrapeado.
    Relación 1:N con Reviews y OwnerResponses.
    """
    __tablename__ = "places"
    __table_args__ = (
        Index('idx_place_id_unique', 'place_id', unique=True),
        Index('idx_name_location', 'name', 'location'),
    )
    
    id = Column(String(50), primary_key=True)
    place_id = Column(String(255), unique=True, nullable=False, index=True)
    name = Column(String(255), nullable=False, index=True)
    location = Column(String(500), nullable=True)
    url = Column(Text, nullable=True)
    phone = Column(String(20), nullable=True)
    
    # Rating aggregado
    rating = Column(Float, nullable=True, default=0.0)
    total_reviews = Column(Integer, nullable=True, default=0)
    
    # Metadata
    created_at = Column(DateTime, default=datetime.utcnow, nullable=False)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow, nullable=False)
    last_scraped = Column(DateTime, nullable=True)
    
    # Relaciones
    reviews: List['Reviews'] = relationship(
        'Reviews', 
        back_populates='place',
        cascade='all, delete-orphan',
        lazy='select'
    )
    kpi_summaries: List['KPISummary'] = relationship(
        'KPISummary',
        back_populates='place',
        cascade='all, delete-orphan',
        lazy='select'
    )
    
    def __repr__(self) -> str:
        return f"<Place(id='{self.id}', name='{self.name}', rating={self.rating})>"
    
    def to_dict(self) -> dict:
        """Convertir objeto a diccionario."""
        return {
            'id': self.id,
            'place_id': self.place_id,
            'name': self.name,
            'location': self.location,
            'url': self.url,
            'phone': self.phone,
            'rating': self.rating,
            'total_reviews': self.total_reviews,
            'created_at': self.created_at.isoformat(),
            'updated_at': self.updated_at.isoformat(),
            'last_scraped': self.last_scraped.isoformat() if self.last_scraped else None
        }


class Reviews(Base):
    """
    Tabla de reseñas individuales.
    
    Almacena cada reseña scrapeada con su metadata y análisis.
    Relación N:1 con Places.
    Relación 1:1 con OwnerResponses (opcional).
    """
    __tablename__ = "reviews"
    __table_args__ = (
        Index('idx_review_place', 'place_id'),
        Index('idx_review_date', 'review_date'),
        Index('idx_review_rating', 'rating'),
        Index('idx_review_sentiment', 'sentiment_score'),
        Index('idx_place_date', 'place_id', 'review_date'),
    )
    
    id = Column(String(50), primary_key=True)
    place_id = Column(String(255), ForeignKey('places.place_id', ondelete='CASCADE'), nullable=False, index=True)
    
    # Información de la reseña
    reviewer_name = Column(String(255), nullable=True)
    reviewer_id = Column(String(255), nullable=True)
    rating = Column(Integer, nullable=False)  # 1-5
    title = Column(String(500), nullable=True)
    text = Column(Text, nullable=False)
    
    # Análisis de texto
    sentiment_score = Column(Float, nullable=True)  # -1.0 a 1.0
    topics = Column(JSON, nullable=True)  # Lista de tópicos detectados
    keywords = Column(JSON, nullable=True)  # Palabras clave extraídas
    
    # Metadata temporal
    review_date = Column(DateTime, nullable=False, index=True)
    scraped_at = Column(DateTime, default=datetime.utcnow, nullable=False)
    
    # Flags de anomalía
    is_anomaly = Column(Boolean, default=False, nullable=False)
    anomaly_type = Column(String(50), nullable=True)  # 'fake', 'extreme', 'spam'
    anomaly_score = Column(Float, nullable=True)  # 0-1
    
    # Relaciones
    place: 'Places' = relationship('Places', back_populates='reviews')
    owner_response: Optional['OwnerResponses'] = relationship(
        'OwnerResponses',
        back_populates='review',
        uselist=False,
        cascade='all, delete-orphan'
    )
    
    def __repr__(self) -> str:
        return f"<Review(id='{self.id}', rating={self.rating}, sentiment={self.sentiment_score})>"
    
    def to_dict(self) -> dict:
        """Convertir objeto a diccionario."""
        return {
            'id': self.id,
            'place_id': self.place_id,
            'reviewer_name': self.reviewer_name,
            'rating': self.rating,
            'title': self.title,
            'text': self.text,
            'sentiment_score': self.sentiment_score,
            'topics': self.topics,
            'review_date': self.review_date.isoformat(),
            'is_anomaly': self.is_anomaly,
            'anomaly_type': self.anomaly_type,
            'anomaly_score': self.anomaly_score
        }


class OwnerResponses(Base):
    """
    Tabla de respuestas del propietario.
    
    Almacena las respuestas del dueño a cada reseña.
    Relación N:1 con Reviews.
    """
    __tablename__ = "owner_responses"
    __table_args__ = (
        Index('idx_response_review', 'review_id'),
        Index('idx_response_date', 'response_date'),
    )
    
    id = Column(String(50), primary_key=True)
    review_id = Column(String(50), ForeignKey('reviews.id', ondelete='CASCADE'), nullable=False, index=True, unique=True)
    
    # Texto de respuesta
    text = Column(Text, nullable=False)
    sentiment_score = Column(Float, nullable=True)  # -1.0 a 1.0
    
    # Análisis temporal
    response_date = Column(DateTime, nullable=False, index=True)
    response_time_hours = Column(Integer, nullable=True)  # Horas hasta responder
    
    # Metadata
    created_at = Column(DateTime, default=datetime.utcnow, nullable=False)
    
    # Relación
    review: 'Reviews' = relationship('Reviews', back_populates='owner_response')
    
    def __repr__(self) -> str:
        return f"<OwnerResponse(id='{self.id}', sentiment={self.sentiment_score})>"
    
    def to_dict(self) -> dict:
        """Convertir objeto a diccionario."""
        return {
            'id': self.id,
            'review_id': self.review_id,
            'text': self.text,
            'sentiment_score': self.sentiment_score,
            'response_time_hours': self.response_time_hours,
            'response_date': self.response_date.isoformat()
        }


class KPISummary(Base):
    """
    Tabla de resúmenes KPI.
    
    Almacena KPIs agregados por local y período.
    Tabla desnormalizada para optimizar queries de lectura.
    """
    __tablename__ = "kpi_summary"
    __table_args__ = (
        Index('idx_kpi_place', 'place_id'),
        Index('idx_kpi_period', 'period_start', 'period_end'),
        Index('idx_kpi_place_period', 'place_id', 'period_start'),
        UniqueConstraint('place_id', 'period_start', 'period_end', name='unique_kpi_period'),
    )
    
    id = Column(String(50), primary_key=True)
    place_id = Column(String(255), ForeignKey('places.place_id', ondelete='CASCADE'), nullable=False, index=True)
    
    # Período del análisis
    period_start = Column(DateTime, nullable=False)
    period_end = Column(DateTime, nullable=False)
    
    # KPIs básicos
    avg_rating = Column(Float, nullable=True)  # Promedio de rating
    total_reviews = Column(Integer, nullable=True)  # Cantidad total
    new_reviews_count = Column(Integer, nullable=True)  # Nuevas en el período
    
    # Análisis de respuestas
    owner_response_count = Column(Integer, nullable=True)  # Respuestas totales
    response_rate = Column(Float, nullable=True)  # Porcentaje respondidas
    avg_response_time_hours = Column(Integer, nullable=True)  # Tiempo promedio respuesta
    
    # Análisis de sentimiento
    avg_sentiment = Column(Float, nullable=True)  # Sentimiento promedio
    positive_count = Column(Integer, nullable=True)  # Reseñas positivas
    neutral_count = Column(Integer, nullable=True)  # Reseñas neutrales
    negative_count = Column(Integer, nullable=True)  # Reseñas negativas
    
    # Análisis temporal
    trend_direction = Column(String(20), nullable=True)  # 'UP', 'DOWN', 'STABLE'
    trend_value = Column(Float, nullable=True)  # Cambio en rating
    week_over_week_growth = Column(Float, nullable=True)  # Crecimiento WoW
    
    # Análisis de anomalías
    anomaly_count = Column(Integer, nullable=True)  # Reseñas anómalas detectadas
    anomaly_rate = Column(Float, nullable=True)  # Porcentaje de anomalías
    
    # Metadata
    calculated_at = Column(DateTime, default=datetime.utcnow, nullable=False)
    valid_until = Column(DateTime, nullable=True)  # Cache expiration
    
    # Relación
    place: 'Places' = relationship('Places', back_populates='kpi_summaries')
    
    def __repr__(self) -> str:
        return f"<KPISummary(place_id='{self.place_id}', period={self.period_start.date()})>"
    
    def to_dict(self) -> dict:
        """Convertir objeto a diccionario."""
        return {
            'id': self.id,
            'place_id': self.place_id,
            'period_start': self.period_start.isoformat(),
            'period_end': self.period_end.isoformat(),
            'avg_rating': self.avg_rating,
            'total_reviews': self.total_reviews,
            'new_reviews_count': self.new_reviews_count,
            'owner_response_count': self.owner_response_count,
            'response_rate': self.response_rate,
            'avg_sentiment': self.avg_sentiment,
            'positive_count': self.positive_count,
            'negative_count': self.negative_count,
            'trend_direction': self.trend_direction,
            'trend_value': self.trend_value,
            'anomaly_count': self.anomaly_count,
            'anomaly_rate': self.anomaly_rate
        }


class AnalysisCache(Base):
    """
    Tabla de caché para análisis.
    
    Almacena resultados de análisis para evitar recalcular.
    Soporta invalidación automática por TTL.
    """
    __tablename__ = "analysis_cache"
    __table_args__ = (
        Index('idx_cache_place', 'place_id'),
        Index('idx_cache_query_type', 'query_type'),
        Index('idx_cache_expires', 'expires_at'),
    )
    
    id = Column(String(50), primary_key=True)
    place_id = Column(String(255), ForeignKey('places.place_id', ondelete='CASCADE'), nullable=False, index=True)
    
    # Tipo de query
    query_type = Column(String(100), nullable=False)  # 'kpis', 'sentiment', 'trends'
    
    # Datos cacheados
    result_json = Column(JSON, nullable=False)
    
    # Control de cache
    cached_at = Column(DateTime, default=datetime.utcnow, nullable=False)
    expires_at = Column(DateTime, nullable=False, index=True)
    
    def is_expired(self) -> bool:
        """Verificar si el cache ha expirado."""
        return datetime.utcnow() > self.expires_at
    
    def __repr__(self) -> str:
        return f"<AnalysisCache(place_id='{self.place_id}', type='{self.query_type}')>"

    # Tipo de predicción
    prediction_type = Column(String(50), nullable=False)  # rating_forecast, volume_forecast, sentiment_forecast
    
    # Fechas del pronóstico
    forecast_start_date = Column(DateTime, nullable=False)
    forecast_end_date = Column(DateTime, nullable=False)
    forecast_period_days = Column(Integer, nullable=True)  # Usually 28 for 4-week forecast
    
    # Pronóstico de rating
    predicted_rating = Column(Float, nullable=True)  # 1.0 a 5.0
    forecast_confidence_lower = Column(Float, nullable=True)  # 95% CI lower bound
    forecast_confidence_upper = Column(Float, nullable=True)  # 95% CI upper bound
    forecast_method = Column(String(50), nullable=True)  # exponential_smoothing, etc
    
    # Pronóstico de volumen
    predicted_volume = Column(Integer, nullable=True)  # Expected reviews count
    
    # Pronóstico de sentimiento
    predicted_sentiment = Column(Float, nullable=True)  # -1.0 a 1.0
    
    # Detección de anomalías
    is_anomaly = Column(Boolean, default=False, nullable=False)
    anomaly_types = Column(JSON, nullable=True)  # Array of anomaly types detected
    anomaly_scores = Column(JSON, nullable=True)  # {anomaly_type: score}

    # Análisis de tendencia
    trend_direction = Column(String(20), nullable=True)  # up, down, stable

    def __repr__(self) -> str:

            'forecast_start_date': self.forecast_start_date.isoformat(),
            'forecast_end_date': self.forecast_end_date.isoformat(),
            'predicted_rating': self.predicted_rating,

class PlaceClusters(Base):
    """"""
    Tabla de asignaciones de clusters y perfiles de lugares.

    Almacena resultados de clustering K-Means (5 perfiles de lugares),
    movimientos de cluster, y patrones de comportamiento detectados.
    Relación N:1 con Places.
    Mantiene histórico de cambios de cluster (snapshots mensuales).
    """"""
    __tablename__ = "place_clusters"
    __table_args__ = (
        Index('idx_cluster_place', 'place_id'),
        Index('idx_cluster_id', 'cluster_id'),
        Index('idx_cluster_label', 'cluster_label'),
        Index('idx_cluster_date', 'calculated_at'),
    )

    id = Column(String(50), primary_key=True)
    place_id = Column(String(255), ForeignKey('places.place_id', ondelete='CASCADE'), nullable=False, index=True)
    
    # Asignación de cluster
    cluster_id = Column(Integer, nullable=False)  # 0-4 representing 5 profiles
    cluster_label = Column(String(50), nullable=False)  # Premium, Good, Standard, In Recovery, Struggling
    cluster_score = Column(Float, nullable=True)  # Distance to cluster center (lower is better)
    
    # Perfil del cluster
    cluster_profile = Column(JSON, nullable=True)  # Complex profile data with characteristics
    
    # Movimiento de cluster
    previous_cluster_id = Column(Integer, nullable=True)
    cluster_movement = Column(String(50), nullable=True)  # stable, promoted, demoted, volatile
    movement_score = Column(Float, nullable=True)  # Magnitude of change
    
    # Patrones de comportamiento detectados
    detected_patterns = Column(JSON, nullable=True)  # Array of {pattern, frequency, sentiment_impact, recommendation}
    
    # Metadata
    calculated_at = Column(DateTime, default=datetime.utcnow, nullable=False, index=True)
    valid_from = Column(DateTime, nullable=False)
    clustering_period = Column(String(100), nullable=True)  # e.g., "2025-02-01 to 2025-02-28"

def get_database_url(env: str = 'sqlite') -> str:
    """
    Obtener URL de conexión a base de datos según ambiente.
    
    Args:
        env: Tipo de ambiente ('sqlite', 'postgresql')
    
    Returns:
        URL de conexión
    """
    import os
    from dotenv import load_dotenv
    
    load_dotenv()
    
    db_url = os.getenv('DATABASE_URL')
    if db_url:
        return db_url
    
    if env == 'sqlite':
        return 'sqlite:///./data/analytics.db'
    elif env == 'postgresql':
        return 'postgresql://user:password@localhost/analytics_db'
    
    return 'sqlite:///./data/analytics.db'


def create_db_engine(database_url: str = None, echo: bool = False):
    """
    Crear motor SQLAlchemy.
    
    Args:
        database_url: URL de conexión
        echo: Si True, mostrar SQL en logs
    
    Returns:
        Motor SQLAlchemy
    """
    if database_url is None:
        database_url = get_database_url()
    
    # Configuración específica para SQLite
    if 'sqlite' in database_url:
        engine = create_engine(
            database_url,
            echo=echo,
            connect_args={'check_same_thread': False},
            poolclass=StaticPool,
        )
    else:
        # PostgreSQL
        engine = create_engine(
            database_url,
            echo=echo,
            pool_pre_ping=True,
            pool_size=10,
            max_overflow=20
        )
    
    return engine


def init_db(database_url: str = None) -> None:
    """
    Inicializar base de datos (crear tablas).
    
    Args:
        database_url: URL de conexión (opcional)
    """
    if database_url is None:
        database_url = get_database_url()
    
    engine = create_db_engine(database_url)
    Base.metadata.create_all(engine)
    print(f"✅ Database initialized at: {database_url}")


def drop_db(database_url: str = None) -> None:
    """
    Eliminar todas las tablas (para testing).
    
    Args:
        database_url: URL de conexión (opcional)
    """
    if database_url is None:
        database_url = get_database_url()
    
    engine = create_db_engine(database_url)
    Base.metadata.drop_all(engine)
    print(f"✅ Database dropped")
